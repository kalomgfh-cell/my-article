# =========================
# 文件名处理
# =========================


def clean_filename(name):


    chars=[
        "\\",
        "/",
        ":",
        "*",
        "?",
        "\"",
        "<",
        ">",
        "|"
    ]


    for c in chars:

        name=name.replace(
            c,
            "-"
        )


    return name.strip()





# =========================
# 提取关键词
# =========================


def get_keyword(title):


    title=title.replace(
        "：",
        "-"
    )



    if "-" in title:

        return title.split("-")[-1].strip()



    return title.strip()
